Option Explicit

Private Sub Workbook_Open()

    ' Open the judge summary sheet.
    On Error Resume Next
    ThisWorkbook.Worksheets(RESULT_SHEET).Activate
    On Error GoTo 0

    ' Arm the automatic start (when AUTO_ARM_ON_OPEN = True in Module_Config).
    '   before AUTO_ARM_TIME : starts automatically at that time
    '   after  AUTO_ARM_TIME : starts right away if JUDGE_END has not passed
    '   after  JUDGE_END     : does nothing
    ArmAutoRunSilent

End Sub

Private Sub Workbook_BeforeClose(Cancel As Boolean)

    ' Make sure the logging loop and the OnTime reservation are stopped.
    StopTickLogging
    DisarmAutoRun

End Sub
